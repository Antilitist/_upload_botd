# Hidden: auto-pair the room, then show the player's match.
jen_blood_night.0009 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "BotD Blood Night auto-pair"
		jen_blood_night_auto_pair_room_effect = yes
		if = {
			limit = { exists = var:jen_blood_night_partner }
			trigger_event = jen_blood_night.0010
		}
		else = {
			trigger_event = jen_blood_night.0020
		}
	}
}
