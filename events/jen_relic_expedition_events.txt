jen_relic_expedition.0021 = {
	type = character_event
	title = jen_relic_expedition.0021.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = jen_relic_fail_kind
					var:jen_relic_fail_kind = flag:junk_fake
				}
				desc = jen_relic_expedition.0021.fake
			}
			triggered_desc = {
				trigger = {
					has_variable = jen_relic_fail_kind
					var:jen_relic_fail_kind = flag:junk_shell
				}
				desc = jen_relic_expedition.0021.shell
			}
			triggered_desc = {
				trigger = {
					has_variable = jen_relic_fail_kind
					var:jen_relic_fail_kind = flag:companion
					exists = scope:jen_dead_companion
				}
				desc = jen_relic_expedition.0021.companion
			}
			triggered_desc = {
				trigger = {
					has_variable = jen_relic_fail_kind
					var:jen_relic_fail_kind = flag:companion
				}
				desc = jen_relic_expedition.0021.companion_generic
			}
			triggered_desc = {
				trigger = {
					has_variable = jen_relic_fail_kind
					var:jen_relic_fail_kind = flag:wound
				}
				desc = jen_relic_expedition.0021.wound
			}
			triggered_desc = {
				trigger = {
					has_variable = jen_relic_fail_kind
					var:jen_relic_fail_kind = flag:curse
				}
				desc = jen_relic_expedition.0021.curse
			}
			desc = jen_relic_expedition.0021.nothing
		}
	}
	theme = jen_dragon_lifestyle
	override_background = { reference = jen_dragon_bg_hall }

	left_portrait = {
		character = root
		animation = worry
	}

	artifact = {
		trigger = {
			exists = scope:jen_new_relic
			has_variable = jen_relic_got_junk
		}
		target = scope:jen_new_relic
		position = lower_center_portrait
	}

	immediate = {
		debug_log = "BotD relic expedition 0021 fail"
	}

	option = {
		name = jen_relic_expedition.0021.a
		remove_variable = jen_relic_got_real
		remove_variable = jen_relic_got_junk
		remove_variable = jen_relic_fail_kind
	}
}
