tradition_jen_keepers_of_the_yard = {
	category = realm

	layers = {
		0 = steward
		1 = western
		4 = jen_keepers_of_the_yard.dds
	}

	is_shown = { jen_culture_can_pick_dragon_tradition_trigger = yes }
	can_pick = {
		custom_tooltip = {
			text = jen_dragon_tradition_requires_blood_tt
			jen_culture_can_pick_dragon_tradition_trigger = yes
		}
	}
	can_pick_for_hybridization = {
		jen_culture_can_pick_dragon_tradition_trigger = yes
	}

	parameters = {
		jen_keepers_of_the_yard = yes
	}
	character_modifier = {
		castle_holding_build_gold_cost = -0.1
		castle_holding_holding_build_gold_cost = -0.1
		monthly_county_control_growth_add = 0.3
	}

	cost = { prestige = { add = { value = tradition_base_cost desc = BASE format = "BASE_VALUE_FORMAT" } multiply = tradition_replacement_cost_if_relevant } }
	ai_will_do = {
		value = 0
		if = { limit = { has_cultural_pillar = heritage_valyrian } add = 80 }
	}
}
