		}

	}

}



jen_dragon_lifestyle_yearly = {

	effect = {

		if = {

			limit = {

				has_perk = jen_dragon_reins_perk

				has_trait = dragonrider

			}

			agot_add_dragon_training_xp = { VALUE = 5 }

		}

		if = {

			limit = {

				has_perk = jen_dragon_pulse_perk

				has_trait = dragonrider

			}

			add_trait_xp = {

				trait = dragonrider

				track = dragon_bond

				value = 5

			}

		}

		if = {

			limit = {

				has_trait = dragonrider

				culture = { has_cultural_tradition = tradition_jen_wings_over_the_wall }

			}

			agot_add_dragon_training_xp = { VALUE = 5 }

		}

		if = {

			limit = {

				has_trait = dragonrider

				culture = { has_cultural_tradition = tradition_jen_fire_made_flesh }

			}

			add_trait_xp = {

				trait = dragonrider

				track = dragon_bond

				value = 5

			}

		}

	}

}



jen_dragon_lifestyle_progress_pulse = {

	trigger = {

		has_lifestyle = jen_dragon_lifestyle

		jen_agot_can_use_dragon_lifestyle_trigger = yes

		is_playable_character = yes

		trigger_if = {

			limit = { is_ai = no }

			has_had_focus_for_days > 30

		}

		NOT = {

			has_character_flag = had_recent_lifestyle_ongoing_event

		}

	}



	first_valid_on_action = {

		jen_dragon_lifestyle_ongoing_events

	}

}



jen_dragon_lifestyle_ongoing_events = {

	trigger = {

		is_available = yes

	}



	random_on_action = {

		chance_to_happen = 10

		chance_of_no_event = {

			if = {

				limit = {

					is_ai = yes

					highest_held_title_tier >= tier_kingdom

				}

				add = 30

			}

			else_if = {

				limit = { is_ai = yes }

				add = 80

			}

		}



		100 = jen_dragon_lifestyle_common_events

	}

}



jen_dragon_lifestyle_common_events = {

	random_events = {

		100 = jen_dragon_lifestyle.1000

		100 = jen_dragon_lifestyle.1010

		100 = jen_dragon_lifestyle.1020

		100 = jen_dragon_lifestyle.1030

	}

}

jen_kin_oath_on_death = {
	trigger = {
		exists = var:jen_oath_kin
		var:jen_oath_kin = { is_alive = yes }
	}
	effect = {
		debug_log = "BotD kin-oath death ping"
		var:jen_oath_kin = { save_scope_as = living_oath }
		save_scope_as = dead_kin
		scope:living_oath = {
			if = {
				limit = { is_ai = no }
				trigger_event = {
					id = jen_dragon_court.1431
					days = { 1 3 }
				}
			}
			else = {
				jen_kin_oath_widow_effect = yes
			}
		}
	}
}
