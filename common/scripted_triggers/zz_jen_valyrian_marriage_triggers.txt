# Blood-Wives: 40/40 and female concubines. No dragonblood gate on marry/concubine.
# These two keys replace AGOT's copies (BotD loads after).

marriage_interaction_valid_target_trigger = {
	trigger_if = {
		limit = {
			scope:secondary_actor = {
				OR = {
					is_clergy = yes
					agot_has_clergy_trait = yes
				}
				NOT = {
					faith = {
						has_doctrine_parameter = clergy_can_marry
					}
				}
			}
		}
		custom_description = {
			text = character_is_clergy_and_cannot_marry
			subject = scope:secondary_actor
			object = scope:secondary_recipient
			scope:secondary_actor = {
				is_clergy = no
				agot_has_clergy_trait = no
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:secondary_recipient = {
				OR = {
					is_clergy = yes
					agot_has_clergy_trait = yes
				}
				NOT = {
					faith = {
						has_doctrine_parameter = clergy_can_marry
					}
				}
			}
		}
		custom_description = {
			text = character_is_clergy_and_cannot_marry
			subject = scope:secondary_recipient
			object = scope:secondary_actor
			scope:secondary_recipient = {
				is_clergy = no
				agot_has_clergy_trait = no
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:secondary_actor = { is_adult = yes }
			scope:secondary_recipient = { is_adult = yes }
		}
		scope:secondary_actor = {
			custom_description = {
				text = "can_marry_check_secondary_actor"
				subject = scope:secondary_actor
				object = scope:secondary_recipient
				can_marry_character_trigger = { CHARACTER = scope:secondary_recipient }
			}
		}
	}
	trigger_else = {
		scope:secondary_actor = {
			custom_description = {
				text = "can_betroth_check_secondary_actor"
				subject = scope:secondary_actor
				object = scope:secondary_recipient
				can_marry_character_trigger = { CHARACTER = scope:secondary_recipient }
			}
		}
	}

	trigger_if = {
		limit = {
			scope:actor = { is_ai = yes }
		}
		NOR = {
			scope:secondary_actor = { has_character_flag = has_been_offered_as_concubine }
			scope:secondary_recipient = { has_character_flag = has_been_offered_as_concubine }
		}
	}

	scope:secondary_recipient = { agot_has_religious_order_trait = no }
	scope:secondary_actor = { agot_has_religious_order_trait = no }
	trigger_if = {
		limit = {
			scope:actor = { is_ai = yes }
			scope:recipient = { is_ai = yes }
		}
		ai_marriage_restrictions_trigger = yes
	}

	# Blood-Wives: no dragonblood gate. Player picks wives vs concubines.
}

make_concubine_is_valid_trigger = {
	custom_description = {
		text = take_concubine_allowed_more
		object = scope:recipient
		scope:actor = {
			jen_has_consort_room_trigger = yes
		}
	}
	scope:recipient = {
		custom_tooltip = {
			text = take_concubine_ex_spouse
			NOT = {
				has_opinion_modifier = {
					modifier = divorced_me_opinion
					target = scope:actor
				}
			}
		}
		OR = {
			AND = {
				is_imprisoned_by = scope:actor
				is_married = yes
			}
			is_married = no
		}
		is_hostage = no
	}

}

jen_has_two_close_kin_spouses_trigger = {
	any_spouse = {
		count >= 2
		is_alive = yes
		is_close_family_of = prev
	}
}

jen_spouse_is_kin_of_root_trigger = {
	OR = {
		is_close_family_of = prev
		is_extended_family_of = prev
	}
}

# Same-sex marriage: default CK3 game rule blocks it even when the faith accepts homosexuality.
# Blood-Wives punches through. Keep AGOT/vanilla path as the first OR branch.
allowed_to_marry_same_sex_trigger = {
	OR = {
		AND = {
			has_game_rule = accepted_same_sex_marriage
			faith = {
				NOR = {
					has_doctrine_parameter = homosexuality_shunned
					has_doctrine_parameter = homosexuality_illegal
				}
			}
		}
		faith = { has_doctrine_parameter = jen_same_sex_marriage }
	}
}

jen_has_consort_room_trigger = {
	OR = {
		allowed_more_concubines = yes
		AND = {
			OR = {
				faith = { has_doctrine_parameter = jen_blood_wives_law }
				faith = { has_doctrine = doctrine_jen_blood_wives }
				faith = { has_doctrine = doctrine_polygamy }
				culture = { has_cultural_parameter = jen_lustful_dragonblood }
			}
			NOT = {
				any_relation = {
					type = jen_consort
					count >= 40
				}
			}
		}
	}
}

make_concubine_is_shown_trigger = {
	scope:actor = {
		is_adult = yes
		OR = {
			allowed_concubines = yes
			faith = { has_doctrine_parameter = jen_blood_wives_law }
			faith = { has_doctrine = doctrine_jen_blood_wives }
			faith = { has_doctrine = doctrine_polygamy }
			culture = { has_cultural_parameter = jen_lustful_dragonblood }
		}
	}
	scope:recipient = {
		OR = {
			is_courtier_of = scope:actor
			is_imprisoned_by = scope:actor
		}
		can_become_concubine_of_character_valid_trigger = { CHARACTER = scope:actor }
	}
}
