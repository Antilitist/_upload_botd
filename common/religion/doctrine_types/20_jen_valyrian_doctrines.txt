	is_shown = {
		OR = {
			religion_tag = jen_valyrian_religion
			has_doctrine = doctrine_jen_blood_wives
		}
	}

	can_pick = {
		custom_tooltip = {
			text = doctrine_jen_blood_wives_religion_tt
			OR = {
				religion_tag = jen_valyrian_religion
				has_doctrine = doctrine_jen_blood_wives
			}
		}
	}

	piety_cost = {
		value = faith_doctrine_cost_mid
		if = {
			limit = { has_doctrine = doctrine_jen_blood_wives }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	parameters = {

		men_can_have_consorts = yes
		women_can_have_consorts = yes
		jen_blood_wives_law = yes
	}
}
