# Dragon lifestyle perk toast, same pattern as Learning/Wanderer.

action_lifestyle_jen_dragon = {
	type = alert
	check_create_action = {
		if = {
			limit = {
				has_focus_jen_dragon = yes
				perk_points > 0
				is_adult = yes
			}
			try_create_important_action = {
				important_action_type = action_lifestyle_jen_dragon
				actor = root
			}
		}
	}

	effect = {
		scope:actor = {
			open_view_data = {
				view = lifestyle
			}
		}
	}
}
