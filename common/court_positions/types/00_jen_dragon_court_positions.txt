jen_dragon_yard_keeper_court_position = {
	sort_order = 990
	max_available_positions = 1
	minimum_rank = county
	skill = stewardship

	court_position_asset = {
		animation = idle
		background = "gfx/interface/illustrations/event_scenes/jen_dragon_cp_yard_keeper.dds"
	}

	opinion = { value = regular_court_position_opinion }
	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 15
		add = {
			value = stewardship
			multiply = 2.5
			max = 50
			desc = court_position_skill_stewardship
		}
		add = {
			value = learning
			multiply = 1.0
			max = 20
			desc = court_position_skill_learning
		}
		if = {
			limit = { has_trait = diligent }
			add = { value = 10 desc = court_position_diligent_trait }
		}
		if = {
			limit = { has_trait = administrator }
			add = { value = 15 desc = court_position_administrator_trait }
		}
		if = {
			limit = { culture ?= { has_cultural_pillar = heritage_valyrian } }
			add = { value = 10 desc = jen_dragon_court_aptitude_valyrian }
		}
		if = {
			limit = { jen_has_education_dragon_trigger = yes }
			add = { value = 15 desc = jen_dragon_court_aptitude_dragon_education }
		}
		if = {
			limit = { has_trait = dragonrider }
			add = { value = 20 desc = jen_dragon_court_aptitude_dragonrider }
		}
		if = {
			limit = { has_character_flag = historical_dragonseed }
			add = { value = 15 desc = jen_dragon_court_aptitude_dragonseed }
		}
		add = court_position_aptitude_high_penalty_value
		add = general_aptitude_bonus
	}

	is_shown = { jen_dragon_court_can_employ_trigger = yes }
	valid_position = {
		jen_dragon_court_can_employ_trigger = yes
		highest_held_title_tier >= tier_county
	}

	is_shown_character = {
		scope:employee = { jen_dragon_court_employee_base_trigger = yes }
	}
	valid_character = {
		scope:employee = {
			jen_dragon_court_employee_base_trigger = yes
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

	revoke_cost = {
		prestige = {
			value = minor_court_position_prestige_revoke_cost
			if = {
				limit = { exists = this is_ai = yes }
				multiply = 0
			}
		}
	}
	salary = {
		prestige = { value = minor_court_position_prestige_salary }
		round = no
	}
	received_salary = {
		prestige = { value = minor_court_position_prestige_salary }
		round = no
	}

	scaling_employer_modifiers = {
		terrible = { monthly_prestige = 0.1 build_gold_cost = -0.02 }
		poor = { monthly_prestige = 0.15 build_gold_cost = -0.03 }
		average = { monthly_prestige = 0.2 build_gold_cost = -0.04 }
		good = { monthly_prestige = 0.25 build_gold_cost = -0.05 }
		excellent = { monthly_prestige = 0.3 build_gold_cost = -0.06 }

	custom_employer_modifier_description = jen_dragon_yard_keeper_employer_custom_effect_description
