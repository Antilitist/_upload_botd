jen_dragon_clutch_perk = {
	lifestyle = jen_dragon_lifestyle
	tree = jen_dragon_raising
	position = { 0 0 }
	icon = node_stewardship

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					employs_court_position = head_dragonkeeper_court_position
					jen_has_education_dragon_trigger = yes
				}
			}
			add = 1989
		}
		if = {
			limit = { has_focus = jen_dragon_raising_focus }
			multiply = 5
		}
		if = {
			limit = { can_start_new_lifestyle_tree_trigger = no }
			multiply = 0
		}
	}

	character_modifier = {
		stewardship = 1
		learning = 1
	}

	effect = {
		custom_description_no_bullet = {
			text = jen_dragon_clutch_perk_effect
		}
	}
}
