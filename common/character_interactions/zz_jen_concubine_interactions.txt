# Blood-Wives: Make Concubine + find_concubine for both sexes. Cap 40. Same-sex already on the tenet.

make_concubine_interaction = {
	category = interaction_category_diplomacy
	icon = concubine_icon

	desc = make_concubine_interaction_desc

	interface = concubine_list

	redirect = {
		scope:actor = {
			save_scope_as = secondary_actor
		}
		scope:recipient = {
			save_scope_as = secondary_recipient
		}
	}

	populate_recipient_list = {
		scope:actor = {
			every_courtier = {
				limit = {
					can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
					NOT = { scope:actor = { has_relation_jen_consort = prev } }
				}
				add_to_list = characters
			}
			every_prisoner = {
				limit = {
					NOT = { is_in_list = characters }
					can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
					NOT = { scope:actor = { has_relation_jen_consort = prev } }
				}
				add_to_list = characters
			}
		}
	}

	is_shown = {
		make_concubine_is_shown_trigger = yes
	}

	is_valid_showing_failures_only = {
		make_concubine_is_valid_trigger = yes
	}

	auto_accept = yes

	on_accept = {
		concubine_on_accept_effect = yes
		jen_ensure_consort_effect = yes

		add_clan_unity_interaction_effect = {
			CHARACTER = scope:actor
			TARGET = scope:recipient
			VALUE = minor_unity_gain
			DESC = clan_unity_concubinage.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}
}

find_concubine = {
	category = interaction_category_diplomacy
	icon = concubine_icon

	desc = find_concubine_desc

	interface = concubine_list

	needs_recipient_to_open = no

	redirect = {
		scope:actor = {
			save_scope_as = secondary_actor
		}
		if = {
			limit = {
				scope:actor != scope:recipient
			}
			scope:recipient = {
				save_scope_as = secondary_recipient
			}
		}
	}

	populate_recipient_list = {
		scope:actor = {
			every_courtier = {
				limit = {
					can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
					NOT = { scope:actor = { has_relation_jen_consort = prev } }
				}
				add_to_list = characters
			}
			every_prisoner = {
				limit = {
					NOT = { is_in_list = characters }
					can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
					NOT = { scope:actor = { has_relation_jen_consort = prev } }
				}
				add_to_list = characters
			}
		}
	}

	is_shown = {
		scope:actor = {
			is_adult = yes
			OR = {
				allowed_concubines = yes
				faith = { has_doctrine_parameter = jen_blood_wives_law }
				faith = { has_doctrine = doctrine_jen_blood_wives }
				faith = { has_doctrine = doctrine_polygamy }
				culture = { has_cultural_parameter = jen_lustful_dragonblood }
			}
			trigger_if = {
				limit = {
					NOT = { exists = scope:secondary_recipient }
				}
				scope:actor = scope:recipient
			}
		}
		trigger_if = {
			limit = {
				scope:actor != scope:recipient
			}
			scope:actor = {
				trigger_if = {
					limit = {
						allowed_to_marry_same_sex_trigger = no
					}
					sex_opposite_of = scope:recipient
				}
				NOT = { relation_with_character_is_incestuous_in_my_faith_trigger = { CHARACTER = scope:recipient } }
			}
			scope:recipient = {
				can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
				NOT = { scope:actor = { has_relation_jen_consort = scope:recipient } }
				is_adult = yes
				OR = {
					is_imprisoned_by = scope:actor
					is_courtier_of = scope:actor
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = take_concubine_allowed_more
			object = scope:recipient
			scope:actor = {
				jen_has_consort_room_trigger = yes
			}
		}
		custom_description = {
			text = take_concubine_available_character_opp
			object = scope:actor
			scope:actor = {
				OR = {
					any_courtier = {
						can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
						NOT = { scope:actor = { has_relation_jen_consort = prev } }
					}
					any_prisoner = {
						can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
						NOT = { scope:actor = { has_relation_jen_consort = prev } }
					}
				}
			}
		}
		trigger_if = {
			limit = {
				scope:actor != scope:recipient
			}
			scope:recipient = {
				can_become_concubine_of_character_trigger = { CHARACTER = scope:actor }
			}
		}
	}

	can_be_picked = {
		is_adult = yes
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}

	on_accept = {
		concubine_on_accept_effect = yes
		jen_ensure_consort_effect = yes

		add_clan_unity_interaction_effect = {
			CHARACTER = scope:actor
			TARGET = scope:recipient
			VALUE = minor_unity_gain
			DESC = clan_unity_concubinage.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}
}
