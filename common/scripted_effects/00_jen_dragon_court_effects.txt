# Pick a living dragon in the player's yard for court-position events.

jen_dragon_save_young_yard_dragon_effect = {
	if = {
		limit = {
			any_relation = {
				type = agot_dragon
				jen_dragon_is_young_yard_dragon_trigger = yes
			}
		}
		random_relation = {
			type = agot_dragon
			limit = { jen_dragon_is_young_yard_dragon_trigger = yes }
			save_scope_as = dragon
		}
	}
	else_if = {
		limit = {
			any_courtier = { jen_dragon_is_young_yard_dragon_trigger = yes }
		}
		random_courtier = {
			limit = { jen_dragon_is_young_yard_dragon_trigger = yes }
			save_scope_as = dragon
		}
	}
}

jen_dragon_fire_yard_pit_debate_effect = {
	if = {
		limit = {
			exists = scope:dragon
			scope:dragon = {
				is_alive = yes
				has_trait = dragon
			}
			any_player = {
				is_ai = no
				is_available = yes
				jen_dragon_belongs_in_shaper_yard_trigger = { DRAGON = scope:dragon }
				OR = {
					employs_court_position = jen_dragon_yard_keeper_court_position
					employs_court_position = jen_dragon_pitmaster_court_position
				}
				NOT = { has_character_flag = jen_recent_yard_pit_debate }
			}
		}
		random_player = {
			limit = {
				is_ai = no
				is_available = yes
				jen_dragon_belongs_in_shaper_yard_trigger = { DRAGON = scope:dragon }
				OR = {
					employs_court_position = jen_dragon_yard_keeper_court_position
					employs_court_position = jen_dragon_pitmaster_court_position
				}
				NOT = { has_character_flag = jen_recent_yard_pit_debate }
			}
			trigger_event = {
				id = jen_dragon_court.2000
				days = { 18 28 }
			}
		}
	}
}

jen_dragon_save_any_yard_dragon_effect = {
	jen_dragon_save_young_yard_dragon_effect = yes
	if = {
		limit = { NOT = { exists = scope:dragon } }
		jen_dragon_save_growing_yard_dragon_effect = yes
	}
	if = {
		limit = { NOT = { exists = scope:dragon } }
		jen_dragon_save_grown_yard_dragon_effect = yes
	}
}
