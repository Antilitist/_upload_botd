jen_valyrian_refresh_sister_wife_effect = {
	if = {
		limit = {
			faith = { has_doctrine = tenet_jen_sister_wife }
			jen_has_two_close_kin_spouses_trigger = yes
		}
		if = {
			limit = { NOT = { has_character_modifier = jen_sister_wife_active } }
			add_character_modifier = { modifier = jen_sister_wife_active }
		}
	}
	else = {
		remove_character_modifier = jen_sister_wife_active
	}
}

jen_valyrian_kin_marriage_piety_effect = {
	if = {
		limit = {
			faith = { has_doctrine_parameter = jen_kin_marriage_piety }
			scope:spouse = {
				OR = {
					is_close_family_of = root
					is_extended_family_of = root
				}
			}
		}
		add_piety = major_piety_gain
		scope:spouse = {
			if = {
				limit = { faith = { has_doctrine_parameter = jen_kin_marriage_piety } }
				add_piety = major_piety_gain
			}
		}
	}
}

# AGOT dragonblood is house-modifier based. Copy it if a dragonblood parent
# would otherwise leave a concubine-born child short.
jen_valyrian_ensure_dragonblood_inheritance_effect = {
	if = {
		limit = {
			is_human = yes
			exists = house
			NOT = { house = { has_house_modifier = dragonrider_house_modifier } }
			OR = {
				real_father ?= { agot_is_dragonblood_character = yes }
				real_mother ?= { agot_is_dragonblood_character = yes }
			}
		}
		if = {
			limit = { real_father.house ?= { has_house_modifier = dragonrider_house_modifier } }
			house = { add_house_modifier = dragonrider_house_modifier }
		}
		else_if = {
			limit = { real_mother.house ?= { has_house_modifier = dragonrider_house_modifier } }
			house = { add_house_modifier = dragonrider_house_modifier }
		}
	}
}
