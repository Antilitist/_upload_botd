jen_dragon_add_perk_if_missing_effect = {
	if = {
		limit = { NOT = { has_perk = $PERK$ } }
		add_perk = $PERK$
	}
}

jen_dragon_cc_fill_capstone_trees_effect = {
	if = {
		limit = {
			OR = {
				has_trait = jen_dragonlord
				has_perk = jen_dragonlord_perk
			}
		}
		debug_log = "BotD CC fill bonding tree"
		jen_dragon_fill_bonding_tree_effect = yes
	}
	if = {
		limit = {
			OR = {
				has_trait = jen_dragon_keeper
				has_perk = jen_dragon_keeper_perk
			}
		}
		debug_log = "BotD CC fill raising tree"
		jen_dragon_fill_raising_tree_effect = yes
	}
	if = {
		limit = {
			OR = {
				has_trait = jen_wyrmrider
				has_perk = jen_wyrmrider_perk
			}
		}
		debug_log = "BotD CC fill riding tree"
		jen_dragon_fill_riding_tree_effect = yes
	}
}
