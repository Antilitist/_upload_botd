jen_relic_prepare_illustrious_effect = {
	# Illustrious = wealth/quality 100
	set_artifact_rarity_illustrious = yes
}

jen_relic_finalize_unique_effect = {
	if = {
		limit = { exists = scope:jen_new_relic }
		scope:jen_new_relic = {
			set_artifact_name = $KEY$
			set_artifact_description = $DESC$
		}
	}
}

jen_relic_mark_created_effect = {
	set_global_variable = {
		name = $KEY$
		value = yes
	}
	set_variable = { name = jen_relic_got_real value = yes }
}

		limit = { exists = scope:jen_new_relic }
		scope:jen_new_relic = {

		}
	}
	set_variable = { name = jen_relic_got_junk value = yes }
	set_variable = { name = jen_relic_fail_kind value = $KIND$ }
}

jen_relic_fix_book_candle_effect = {
	hidden_effect = {
		if = {
			limit = {
				any_character_artifact = {
					OR = {
						has_variable = jen_is_glass_candle
						has_artifact_modifier = jen_art_glass_candle_modifier
					}
					NOT = { artifact_type = glass_candle }
				}
			}
			random_character_artifact = {
				limit = {
					OR = {
						has_variable = jen_is_glass_candle
						has_artifact_modifier = jen_art_glass_candle_modifier
					}
					NOT = { artifact_type = glass_candle }
				}
				save_scope_as = old_candle
			}
			if = {
				limit = { exists = scope:old_candle }
				debug_log = "BotD converting book glass candle to AGOT candle visual"
				destroy_artifact = scope:old_candle
				jen_relic_create_glass_candle_effect = yes
			}
		}
	}
}

jen_relic_kill_companion_effect = {
	if = {
		limit = {
			any_knight = {
				is_alive = yes
				is_adult = yes
			}
		}
		random_knight = {
			limit = {
				is_alive = yes
				is_adult = yes
			}
			save_scope_as = jen_dead_companion
			death = {
				death_reason = death_accident
			}
		}
		set_variable = { name = jen_relic_fail_kind value = flag:companion }
	}
	else = {
		stress_impact = {
			base = medium_stress_impact_gain
		}
		set_variable = { name = jen_relic_fail_kind value = flag:nothing }
	}
}

jen_relic_fail_effect = {
	random_list = {
		25 = {
			add_trait = wounded_1
			set_variable = { name = jen_relic_fail_kind value = flag:wound }
		}
		20 = {
			jen_relic_kill_companion_effect = yes
		}
		20 = {
			jen_relic_create_fake_trinket_effect = yes
		}
		20 = {
			add_character_modifier = {
				modifier = jen_relic_sea_curse_modifier
				years = 5
			}
			set_variable = { name = jen_relic_fail_kind value = flag:curse }
		}
		15 = {
			jen_relic_create_empty_shell_effect = yes
		}
	}
	if = {
		limit = {
			var:jen_relic_site = 3
			any_knight = {
				is_alive = yes
				is_adult = yes
				NOT = { jen_faith_is_valyrian_trigger = yes }
			}
		}
		random = {
			chance = 40
			random_knight = {
				limit = {
					is_alive = yes
					is_adult = yes
					NOT = { jen_faith_is_valyrian_trigger = yes }
				}
				death = {
					death_reason = death_accident
				}
			}
		}
	}
	else_if = {
		limit = {
			any_knight = {
				is_alive = yes
				is_adult = yes
				NOT = { jen_faith_is_valyrian_trigger = yes }
			}
		}
		random = {
			chance = 20
			stress_impact = {
			base = medium_stress_impact_gain
		}
		}
	}
}

jen_relic_resolve_expedition_effect = {
	remove_variable = jen_relic_got_real
	remove_variable = jen_relic_got_junk
	remove_variable = jen_relic_fail_kind
	random_list = {
		40 = {
			modifier = {
				add = 20
				has_trait = dragonrider
			}
			modifier = {
				add = 10
				agot_is_dragonblood_character = yes
			}
			modifier = {
				add = 15
				var:jen_relic_site = 4
				intrigue >= 12
			}
			modifier = {
				add = -15
				var:jen_relic_site = 3
			}
			jen_relic_success_effect = yes
		}
		60 = {
			modifier = {
				add = 15
				var:jen_relic_site = 3
			}
			jen_relic_fail_effect = yes
		}
	}
	remove_variable = jen_relic_site
}

jen_relic_open_result_event_effect = {
	jen_relic_resolve_expedition_effect = yes
	if = {
		limit = {
			has_variable = jen_relic_got_real
			exists = scope:jen_new_relic
		}
		debug_log = "BotD relic expedition SUCCESS named find"
		trigger_event = jen_relic_expedition.0020
	}
	else = {
		debug_log = "BotD relic expedition FAIL or junk"
		trigger_event = jen_relic_expedition.0021
	}
}
