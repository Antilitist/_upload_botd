# BotD extra dragon council seats. zz_ load order. CoS council pattern (auto_fill no, fire/reassign, got/fired effects).

jen_dragon_councillor_master_of_dragons = {
	skill = martial
	auto_fill = no
	can_fire = yes
	can_reassign = yes

	name = jen_dragon_councillor_master_of_dragons
	tooltip = jen_dragon_councillor_master_of_dragons_desc

	valid_position = {
		jen_dragon_council_blood_trigger = yes
	}

	valid_character = {
		exists = root.liege_or_court_owner
		jen_dragon_council_blood_trigger = yes
		trigger_if = {
			limit = { NOT = { has_council_position = jen_dragon_councillor_master_of_dragons } }
			NOR = {
				has_council_position = jen_dragon_councillor_lord_of_the_pit
				has_council_position = jen_dragon_councillor_blood_chamberlain
				has_council_position = jen_dragon_councillor_voice_of_the_fourteen
			}
		}
	}

	modifier = {
		name = jen_dragon_councillor_master_of_dragons_modifier
		monthly_martial_lifestyle_xp_gain_mult = 0.1
		prowess = 1
		scale = council_scaled_by_liege_tier
	}

	council_owner_modifier = {
		name = jen_dragon_councillor_master_of_dragons_owner_modifier
		knight_effectiveness_mult = 0.05
		dread_baseline_add = 2
		scale = council_scaled_by_liege_tier
	}

	on_get_position = {
		got_council_position_effect = yes
	}

	on_fired_from_position = {
		fired_from_council_position_effect = yes
	}

	portrait_animation = marshal

	barbershop_data = {
		position = { 80.0 520.0 }
		click_to_front = no
	}
}
